queue = []
MAX = 5

while True:
    print("\n--- PARKING SYSTEM ---")
    print("1. Enqueue Car")
    print("2. Dequeue Car")
    print("3. Display Cars")
    print("4. Exit")

    choice = int(input("Enter your choice: "))

    if choice == 1:
        if len(queue) == MAX:
            print("Parking is full")
        else:
            car = input("Enter car number: ")
            queue.append(car)
            print("Car entered the parking system.")

    elif choice == 2:
        if len(queue) == 0:
            print("Parking is empty")
        else:
            car = queue.pop(0)
            print("Car left the parking system:", car)

    elif choice == 3:
        if len(queue) == 0:
            print("No cars in parking")
        else:
            print("Cars in parking:")
            for car in queue:
                print(car)

    elif choice == 4:
        print("Exit")
        break

    else:
        print("Invalid choice")
