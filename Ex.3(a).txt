stack = []
MAX = 5

while True:
    print("\n--- BOOK STACK ---")
    print("1. Push Book")
    print("2. Pop Book")
    print("3. Display Stack")
    print("4. Exit")

    choice = int(input("Enter your choice: "))

    if choice == 1:
        if len(stack) == MAX:
            print("Stack Overflow")
        else:
            book = input("Enter book title: ")
            stack.append(book)
            print("Book added to stack.")

    elif choice == 2:
        if len(stack) == 0:
            print("Stack Underflow")
        else:
            book = stack.pop()
            print("Removed book:", book)

    elif choice == 3:
        if len(stack) == 0:
            print("Stack is empty")
        else:
            print("Books in Stack:")
            for book in reversed(stack):
                print(book)

    elif choice == 4:
        print("Exit")
        break

    else:
        print("Invalid choice")
